
# ============================================================
# ZRCE VERSION 3.3 TEST
# SPY 1-MINUTE CHART
#
# FIXED SIMULATOR RECORD
# EXACT INTEGER-CENT STOP/TARGET STORAGE
# SEPARATE LOCKED ENTRY REGISTER
# EXPLICIT ENTRY REJECTION DIAGNOSTICS
#
# MAJOR ZONES V4.9 IS A SEPARATE STUDY
# ============================================================

declare upper;

# ============================================================
# INPUTS
# ============================================================

input zoneSize = 0.05;
input meetingTolerance = 0.05;
input cooldownMinutes = 10;

input showZ = yes;
input showR = yes;
input showC = yes;
input showE = yes;

input showZones = yes;
input showStatus = yes;
input showEntryDiagnostic = yes;
input showProfitTargets = yes;
input showTPDiagnostics = yes;

input showMockTrades = yes;
input showMockStats = yes;
input showSimDiagnostics = yes;
input showHistoricalDiagnostics = yes;

# ============================================================
# SESSION
# ============================================================

def bn = BarNumber();
def dayNumber = GetYYYYMMDD();

def newDay =
    bn == 1 or
    dayNumber != dayNumber[1];

def inSession =
    SecondsFromTime(0930) >= 0 and
    SecondsTillTime(1600) > 0;

def minutesFromOpen =
    Floor(SecondsFromTime(0930) / 60);

def fiveMinuteBoundary =
    inSession and
    minutesFromOpen >= 10 and
    minutesFromOpen % 5 == 0;

# ============================================================
# COMPLETED FIVE-MINUTE CANDLES
# ============================================================

def firstOpen = open[10];
def firstClose = close[6];

def secondOpen = open[5];
def secondClose = close[1];

def validPair =
    fiveMinuteBoundary and
    dayNumber == dayNumber[10];

def demandPattern =
    validPair and
    firstClose < firstOpen and
    secondClose > secondOpen and
    AbsValue(firstClose - secondOpen)
        <= meetingTolerance;

def supplyPattern =
    validPair and
    firstClose > firstOpen and
    secondClose < secondOpen and
    AbsValue(firstClose - secondOpen)
        <= meetingTolerance;

def demandCandidate =
    Min(firstClose, secondOpen);

def supplyCandidate =
    Max(firstClose, secondOpen);

# ============================================================
# MAJOR DEMAND AND SUPPLY LEVELS
# ============================================================

def majorDemand = CompoundValue(
    1,
    if newDay then 0
    else if demandPattern then
        if majorDemand[1] == 0
        then demandCandidate
        else Min(majorDemand[1], demandCandidate)
    else majorDemand[1],
    0
);

def majorSupply = CompoundValue(
    1,
    if newDay then 0
    else if supplyPattern then
        if majorSupply[1] == 0
        then supplyCandidate
        else Max(majorSupply[1], supplyCandidate)
    else majorSupply[1],
    0
);

def demandBottom = majorDemand;
def demandTop = majorDemand + zoneSize;

def supplyTop = majorSupply;
def supplyBottom = majorSupply - zoneSize;

# ============================================================
# ZONE DISPLAY
# ============================================================

plot DemandZoneTop =
    if showZones and inSession and majorDemand > 0
    then demandTop
    else Double.NaN;

plot DemandZoneBottom =
    if showZones and inSession and majorDemand > 0
    then demandBottom
    else Double.NaN;

plot SupplyZoneTop =
    if showZones and inSession and majorSupply > 0
    then supplyTop
    else Double.NaN;

plot SupplyZoneBottom =
    if showZones and inSession and majorSupply > 0
    then supplyBottom
    else Double.NaN;

DemandZoneTop.SetDefaultColor(Color.GREEN);
DemandZoneBottom.SetDefaultColor(Color.GREEN);
SupplyZoneTop.SetDefaultColor(Color.RED);
SupplyZoneBottom.SetDefaultColor(Color.RED);

DemandZoneTop.SetLineWeight(1);
DemandZoneBottom.SetLineWeight(1);
SupplyZoneTop.SetLineWeight(1);
SupplyZoneBottom.SetLineWeight(1);

AddCloud(
    DemandZoneTop,
    DemandZoneBottom,
    Color.DARK_GREEN,
    Color.DARK_GREEN
);

AddCloud(
    SupplyZoneTop,
    SupplyZoneBottom,
    Color.DARK_RED,
    Color.DARK_RED
);

# ============================================================
# Z CANDLE CONDITIONS
# ============================================================

def rawLongZ =
    inSession and
    majorDemand > 0 and
    close < open and
    open > demandTop and
    low <= demandTop;

def rawShortZ =
    inSession and
    majorSupply > 0 and
    close > open and
    open < supplyBottom and
    high >= supplyBottom;

# ============================================================
# ZRCE STATE MACHINE
# ============================================================

def state = CompoundValue(
    1,

    if newDay or !inSession then 0

    else if state[1] >= 10 then
        if state[1] == 10
        then 0
        else state[1] - 1

    else if state[1] == 1 then
        if close > open
        then 2
        else 10 + cooldownMinutes

    else if state[1] == 2 then
        if close > open and
           close > close[1] and
           close > demandTop[2]
        then 3
        else 10 + cooldownMinutes

    else if state[1] == 3 then 0

    else if state[1] == -1 then
        if close < open
        then -2
        else 10 + cooldownMinutes

    else if state[1] == -2 then
        if close < open and
           close < close[1] and
           close < supplyBottom[2]
        then -3
        else 10 + cooldownMinutes

    else if state[1] == -3 then 0

    else if rawLongZ then 1
    else if rawShortZ then -1

    else 0,

    0
);

# ============================================================
# Z / R / C / E FLAGS
# ============================================================

def longZ =
    state == 1 and state[1] == 0;

def shortZ =
    state == -1 and state[1] == 0;

def longR =
    state == 2 and state[1] == 1;

def shortR =
    state == -2 and state[1] == -1;

def longC =
    state == 3 and state[1] == 2;

def shortC =
    state == -3 and state[1] == -2;

def longE =
    inSession and state[1] == 3;

def shortE =
    inSession and state[1] == -3;

# ============================================================
# ENTRY DIRECTION
# ============================================================

def entryActive =
    longE or shortE;

def confirmationClose =
    if entryActive
    then close[1]
    else Double.NaN;

def entryOpen =
    if entryActive
    then open
    else Double.NaN;

def entryDifference =
    if entryActive
    then entryOpen - confirmationClose
    else Double.NaN;

def longDirectionOK =
    longE and open >= close[1];

def shortDirectionOK =
    shortE and open <= close[1];

def directionOK =
    longDirectionOK or shortDirectionOK;

def cooldownRemaining =
    if state >= 10
    then state - 10
    else 0;

# ============================================================
# ZRCE SIGNAL BUBBLES
# ============================================================

AddChartBubble(
    showZ and longZ,
    low,
    "Z",
    Color.GREEN,
    no
);

AddChartBubble(

    low,
    "R",
    Color.GREEN,
    no
);

AddChartBubble(
    showC and longC,
    low,
    "C",
    Color.GREEN,
    no
);

AddChartBubble(
    showE and longDirectionOK,
    open,

    Color.CYAN,
    no
);

AddChartBubble(
    showZ and shortZ,
    high,
    "Z",
    Color.RED,
    yes
);

AddChartBubble(
    showR and shortR,
    high,
    "R",
    Color.RED,
    yes
);

AddChartBubble(
    showC and shortC,
    high,
    "C",
    Color.RED,
    yes
);

AddChartBubble(
    showE and shortDirectionOK,
    open,
    "E SHORT",
    Color.MAGENTA,
    yes
);

# ============================================================
# ENTRY ARROWS
# ============================================================

plot LongEntry =
    if showE and longDirectionOK
    then open
    else Double.NaN;

    PaintingStrategy.ARROW_UP

plot ShortEntry =

    then open
    else Double.NaN;

    PaintingStrategy.ARROW_DOWN
);
ShortEntry.SetDefaultColor(Color.MAGENTA);
ShortEntry.SetLineWeight(3);

# ============================================================
# FIVE-MINUTE PROFIT TARGETS
# ============================================================

def ptFiveOpen = open[5];
def ptFiveClose = close[1];

def ptFiveLow = Lowest(low[1], 5);
def ptFiveHigh = Highest(high[1], 5);

def ptComplete =
    fiveMinuteBoundary and
    dayNumber == dayNumber[5];

def ptDemandNew =
    ptComplete and
    demandPattern and
    (
        majorDemand[1] == 0 or
        demandCandidate < majorDemand[1]
    );

def ptSupplyNew =
    ptComplete and
    supplyPattern and
    (
        majorSupply[1] == 0 or
        supplyCandidate > majorSupply[1]
    );

def ptDemandTouch =
    ptComplete and
    !ptDemandNew and
    majorDemand[1] > 0 and
    ptFiveClose < ptFiveOpen and
    ptFiveLow <= majorDemand[1] + zoneSize and
    ptFiveHigh >= majorDemand[1];

def ptSupplyTouch =
    ptComplete and
    !ptSupplyNew and
    majorSupply[1] > 0 and
    ptFiveClose > ptFiveOpen and
    ptFiveHigh >= majorSupply[1] - zoneSize and
    ptFiveLow <= majorSupply[1];

# ============================================================
# TARGET PHASES
# ============================================================

def ptDemandPhase = CompoundValue(
    1,

    if newDay then 0
    else if ptDemandNew then 2
    else if ptDemandTouch then 1
    else if !ptComplete then ptDemandPhase[1]

    else if
        (ptDemandPhase[1] == 1 or
         ptDemandPhase[1] == 2) and
        ptFiveClose > ptFiveOpen
    then 2

    else 0,

    0
);

def ptSupplyPhase = CompoundValue(
    1,

    if newDay then 0
    else if ptSupplyNew then 2
    else if ptSupplyTouch then 1
    else if !ptComplete then ptSupplyPhase[1]

    else if
        (ptSupplyPhase[1] == 1 or
         ptSupplyPhase[1] == 2) and
        ptFiveClose < ptFiveOpen
    then 2

    else 0,

    0
);

def ptDemandGreen =
    ptComplete and
    !ptDemandNew and
    !ptDemandTouch and
    (
        ptDemandPhase[1] == 1 or
        ptDemandPhase[1] == 2
    ) and
    ptFiveClose > ptFiveOpen;

def ptSupplyRed =
    ptComplete and
    !ptSupplyNew and
    !ptSupplyTouch and
    (
        ptSupplyPhase[1] == 1 or
        ptSupplyPhase[1] == 2
    ) and
    ptFiveClose < ptFiveOpen;

# ============================================================
# TARGET RUN CLOSES
# ============================================================

def ptDemandRunClose = CompoundValue(
    1,

    if newDay then Double.NaN
    else if ptDemandNew then ptFiveClose
    else if ptDemandTouch then Double.NaN
    else if ptDemandGreen then ptFiveClose
    else ptDemandRunClose[1],

    Double.NaN
);

def ptSupplyRunClose = CompoundValue(
    1,

    if newDay then Double.NaN
    else if ptSupplyNew then ptFiveClose
    else if ptSupplyTouch then Double.NaN
    else if ptSupplyRed then ptFiveClose
    else ptSupplyRunClose[1],

    Double.NaN
);

# ============================================================
# COMPLETED TARGETS
# ============================================================

def ptDemandCompleted =
    ptComplete and
    !ptDemandNew and
    !ptDemandTouch and
    ptDemandPhase[1] == 2 and
    ptFiveClose <= ptFiveOpen;

def ptSupplyCompleted =
    ptComplete and
    !ptSupplyNew and
    !ptSupplyTouch and
    ptSupplyPhase[1] == 2 and
    ptFiveClose >= ptFiveOpen;

def ptLastDemandTarget = CompoundValue(
    1,

    if newDay or ptDemandNew
    then Double.NaN

    else if ptDemandCompleted
    then ptDemandRunClose[1]

    else ptLastDemandTarget[1],

    Double.NaN
);

def ptLastSupplyTarget = CompoundValue(
    1,

    if newDay or ptSupplyNew
    then Double.NaN

    else if ptSupplyCompleted
    then ptSupplyRunClose[1]

    else ptLastSupplyTarget[1],

    Double.NaN
);

# ============================================================
# VERSION 3.3 MOCK TRADE SIMULATOR
#
# TRADE RECORD CONTAINS STOP AND TARGET ONLY.
#
# STOP CENTS * 100000 + TARGET CENTS
#
# POSITIVE RECORD = LONG
# NEGATIVE RECORD = SHORT
# ZERO = FLAT
#
# EACH COMPONENT IS AN INTEGER BELOW 100000.
# THE FULL RECORD IS BELOW 10 BILLION,
# WELL WITHIN EXACT INTEGER PRECISION.
#
# ENTRY PRICE IS LOCKED IN A SEPARATE
# REGISTER AFTER THE TRADE OPENS.
# ============================================================

# ============================================================
# STOP CANDIDATES
# ============================================================

def simLongStopCandidate =
    Lowest(low[1], 3);

def simShortStopCandidate =
    Highest(high[1], 3);

# ============================================================
# SAFE CENT VALUES
# ============================================================

def simOpenCents =
    if IsNaN(open)
    then 0
    else Round(open * 100, 0);

def simLongStopCents =
    if IsNaN(simLongStopCandidate)
    then 0
    else Round(simLongStopCandidate * 100, 0);

def simShortStopCents =
    if IsNaN(simShortStopCandidate)
    then 0
    else Round(simShortStopCandidate * 100, 0);

def simLongTargetCents =
    if IsNaN(ptLastDemandTarget)
    then 0
    else Round(ptLastDemandTarget * 100, 0);

def simShortTargetCents =
    if IsNaN(ptLastSupplyTarget)
    then 0
    else Round(ptLastSupplyTarget * 100, 0);

# ============================================================
# VALIDATION
# ============================================================

def simLongTargetValid =
    simLongTargetCents > simOpenCents;

def simShortTargetValid =
    simShortTargetCents > 0 and
    simShortTargetCents < simOpenCents;

def simLongStopValid =
    simLongStopCents > 0 and
    simLongStopCents < simOpenCents;

def simShortStopValid =
    simShortStopCents > simOpenCents;

def simLongPricesValid =
    simOpenCents > 0 and
    simOpenCents < 100000 and
    simLongStopCents > 0 and
    simLongStopCents < 100000 and
    simLongTargetCents > 0 and
    simLongTargetCents < 100000;
def simShortPricesValid =
    simOpenCents > 0 and
    simOpenCents < 100000 and
    simShortStopCents > 0 and
    simShortStopCents < 100000 and
    simShortTargetCents > 0 and
    simShortTargetCents < 100000;

def simLongCandidate =
    longDirectionOK and
    simLongTargetValid and
    simLongStopValid and
    simLongPricesValid;

def simShortCandidate =
    shortDirectionOK and
    simShortTargetValid and
    simShortStopValid and
    simShortPricesValid;

def simClosingBar =
    SecondsFromTime(1559) >= 0 and
    SecondsTillTime(1600) > 0;

# ============================================================
# PACK ONLY STOP AND TARGET
# ============================================================

def simNewLongRecord =
    simLongStopCents * 100000 +
    simLongTargetCents;

def simNewShortRecord =
    simShortStopCents * 100000 +
    simShortTargetCents;

# ============================================================
# SINGLE RECURSIVE POSITION RECORD
#
# NO FORWARD REFERENCES.
# PREVIOUS RECORD CONTAINS THE LOCKED
# STOP AND TARGET.
# ============================================================

def simRecord = CompoundValue(
    1,

    if newDay then 0

    else if IsNaN(simRecord[1]) then 0

    else if simRecord[1] > 0 then

        if
            low <=
                Floor(
                    simRecord[1] / 100000
                ) / 100

            or high >=
                (
                    AbsValue(simRecord[1]) -
                    Floor(
                        AbsValue(simRecord[1]) /
                        100000
                    ) * 100000
                ) / 100

            or simClosingBar
            or !inSession

        then 0
        else simRecord[1]

    else if simRecord[1] < 0 then

        if
            high >=
                Floor(
                    AbsValue(simRecord[1]) /
                    100000
                ) / 100

            or low <=
                (
                    AbsValue(simRecord[1]) -
                    Floor(
                        AbsValue(simRecord[1]) /
                        100000
                    ) * 100000
                ) / 100

            or simClosingBar
            or !inSession

        then 0
        else simRecord[1]

    else if simLongCandidate and
            inSession and
            !simClosingBar
    then simNewLongRecord

    else if simShortCandidate and
            inSession and
            !simClosingBar
    then -simNewShortRecord

    else 0,

    0
);

# ============================================================
# POSITION AND TRANSITIONS
# ============================================================

def simPriorRecord =
    if bn <= 1 or IsNaN(simRecord[1])
    then 0
    else simRecord[1];

def simPosition =
    if simRecord > 0 then 1
    else if simRecord < 0 then -1
    else 0;

def simPriorPosition =
    if simPriorRecord > 0 then 1
    else if simPriorRecord < 0 then -1
    else 0;

def simBuy =
    simPosition == 1 and
    simPriorPosition == 0;

def simSell =
    simPosition == -1 and
    simPriorPosition == 0;

def simExit =
    !newDay and
    simPriorPosition != 0 and
    simPosition == 0;

# ============================================================
# LOCKED ENTRY PRICE
#
# THIS REGISTER IS DECLARED AFTER simRecord
# SO IT CANNOT CAUSE A FORWARD REFERENCE.
# ============================================================

def simEntryRegister = CompoundValue(
    1,

    if newDay then 0

    else if simBuy or simSell
    then open

    else if IsNaN(simEntryRegister[1])
    then 0

    else simEntryRegister[1],

    0
);

# ============================================================
# DECODE CURRENT STOP AND TARGET
# ============================================================

def simPacked =
    AbsValue(simRecord);

def simStopCents =
    Floor(simPacked / 100000);

def simTargetCents =
    simPacked -
    simStopCents * 100000;

def simStop =
    simStopCents / 100;

def simTarget =
    simTargetCents / 100;

def simEntry =
    simEntryRegister;

# ============================================================
# DECODE PREVIOUS STOP AND TARGET
# ============================================================

def simPriorPacked =
    AbsValue(simPriorRecord);

def simPriorStopCents =
    Floor(simPriorPacked / 100000);

def simPriorTargetCents =
    simPriorPacked -
    simPriorStopCents * 100000;

def simPriorStop =
    simPriorStopCents / 100;

def simPriorTarget =
    simPriorTargetCents / 100;

def simPriorEntry =
    if bn <= 1 or IsNaN(simEntryRegister[1])
    then 0
    else simEntryRegister[1];

# ============================================================
# EXIT CLASSIFICATION
#
# STOP HAS PRIORITY IF BOTH ARE HIT.
# ============================================================

def simStopHit =
    simExit and
    (
        (
            simPriorPosition == 1 and
            low <= simPriorStop
        )
        or
        (
            simPriorPosition == -1 and
            high >= simPriorStop
        )
    );

def simTargetHit =
    simExit and
    !simStopHit and
    (
        (
            simPriorPosition == 1 and
            high >= simPriorTarget
        )
        or
        (
            simPriorPosition == -1 and
            low <= simPriorTarget
        )
    );

# ============================================================
# EXIT PRICE
# ============================================================

def simExitPrice =
    if !simExit then 0

    else if simStopHit
    then simPriorStop

    else if simTargetHit
    then simPriorTarget

    else if simClosingBar
    then close

    else if !inSession
    then close[1]

    else close;

# ============================================================
# ONE-SHARE P/L
# ============================================================

def simTradePL =
    if !simExit then 0

    else if simPriorPosition == 1
    then simExitPrice - simPriorEntry

    else if simPriorPosition == -1
    then simPriorEntry - simExitPrice

    else 0;

def simSafeTradePL =
    if IsNaN(simTradePL)
    then 0
    else simTradePL;

# ============================================================
# DAILY STATISTICS
# ============================================================

def simDailyPL = CompoundValue(
    1,

    if newDay then 0

    else
        (
            if IsNaN(simDailyPL[1])
            then 0
            else simDailyPL[1]
        ) + simSafeTradePL,

    0
);

def simDailyTrades = CompoundValue(
    1,

    if newDay then 0

    else
        (
            if IsNaN(simDailyTrades[1])
            then 0
            else simDailyTrades[1]
        ) +
        (if simExit then 1 else 0),

    0
);

def simDailyWins = CompoundValue(
    1,

    if newDay then 0

    else
        (
            if IsNaN(simDailyWins[1])
            then 0
            else simDailyWins[1]
        ) +
        (
            if simExit and simSafeTradePL > 0
            then 1
            else 0
        ),

    0
);

def simDailyLosses = CompoundValue(
    1,

    if newDay then 0

    else
        (
            if IsNaN(simDailyLosses[1])
            then 0
            else simDailyLosses[1]
        ) +
        (
            if simExit and simSafeTradePL < 0
            then 1
            else 0
        ),

    0
);

# ENTRY REJECTION DIAGNOSTICS
# ============================================================

def simRejectedDirection =
    entryActive and
    !directionOK;

def simRejectedTarget =
    directionOK and
    (
        (
            longDirectionOK and
            !simLongTargetValid
        )
        or
        (
            shortDirectionOK and
            !simShortTargetValid
        )
    );

def simRejectedStop =
    directionOK and
    !simRejectedTarget and
    (
        (
            longDirectionOK and
            !simLongStopValid
        )
        or
        (
            shortDirectionOK and
            !simShortStopValid
        )
    );

def simRejectedPriceRange =
    directionOK and
    !simRejectedTarget and
    !simRejectedStop and
    (
        (
            longDirectionOK and
            !simLongPricesValid
        )
        or
        (
            shortDirectionOK and
            !simShortPricesValid
        )
    );

def simRejectedBusy =
    directionOK and
    !simRejectedTarget and
    !simRejectedStop and
    !simRejectedPriceRange and
    simPriorPosition != 0;

def simRejectedSession =
    directionOK and
    !simRejectedTarget and
    !simRejectedStop and
    !simRejectedPriceRange and
    !simRejectedBusy and
    simClosingBar;

def simUnexpectedNoEntry =
    directionOK and
    !simRejectedTarget and
    !simRejectedStop and
    !simRejectedPriceRange and
    !simRejectedBusy and
    !simRejectedSession and
    !simBuy and
    !simSell;
